-- ===================================================================== -- MAHYRA RELAKSIA — 22. SKEMA GAJI DI PORTAL TERAPIS & ABSEN HARIAN -- -- 1. Terapis boleh membaca skema gaji yang berlaku untuknya, supaya -- halaman Komisi menampilkan angka yang benar sesuai jabatan -- dan cabang tempatnya bekerja. -- -- 2. Hari kerja tetap dihitung meski tidak ada orderan. Terapis yang -- datang dan absen tetap memperoleh gaji harian. Karena itu perlu -- absen harian mandiri, tidak selalu menempel pada orderan. -- -- Aman dijalankan berulang kali. -- ===================================================================== -- --------------------------------------------------------------------- -- BAGIAN 1 — Terapis membaca skema gajinya sendiri -- --------------------------------------------------------------------- drop policy if exists sel_skema_gaji on skema_gaji; create policy sel_skema_gaji on skema_gaji for select to authenticated using ( jabatan_id = my_jabatan_id() -- skema untuk jabatan saya or (is_office() and boleh_modul('skema_gaji','lihat')) ); -- --------------------------------------------------------------------- -- BAGIAN 2 — Absen harian tanpa orderan -- --------------------------------------------------------------------- create or replace function absen_harian(p jsonb) returns absen_log language plpgsql security definer set search_path = public as $$ declare v_kar uuid; v_tipe text; hasil absen_log; begin v_kar := my_karyawan_id(); if v_kar is null then raise exception 'Akun ini belum tertaut ke data karyawan.'; end if; v_tipe := coalesce(p->>'tipe','masuk'); if v_tipe not in ('masuk','pulang') then raise exception 'Jenis absen harus masuk atau pulang.'; end if; insert into absen_log (karyawan_id, pesanan_item_id, cabang_id, tanggal, tipe, waktu, foto, lat, lng, alamat, akurasi_meter, catatan) values (v_kar, null, (select cabang_id from karyawan where id = v_kar), current_date, v_tipe, now(), nullif(p->>'foto',''), nullif(p->>'lat','')::numeric, nullif(p->>'lng','')::numeric, nullif(p->>'alamat',''), nullif(p->>'akurasi','')::int, 'Absen harian') returning * into hasil; return hasil; end $$; revoke all on function absen_harian(jsonb) from public; grant execute on function absen_harian(jsonb) to authenticated; -- --------------------------------------------------------------------- -- BAGIAN 3 — Hari kerja efektif: kehadiran dihitung meski tanpa orderan -- --------------------------------------------------------------------- create or replace function hari_kerja_efektif(p_karyawan uuid, p_periode text) returns int language sql stable security definer set search_path = public as $$ select count(*)::int from ( -- hari dengan catatan absensi select tanggal from absensi where karyawan_id = p_karyawan and to_char(tanggal,'YYYY-MM') = p_periode and status in ('hadir','telat') union -- hari dengan check in, termasuk absen harian tanpa orderan select tanggal from absen_log where karyawan_id = p_karyawan and to_char(tanggal,'YYYY-MM') = p_periode union -- hari dengan orderan selesai select date(i.jam_selesai) from pesanan_item i where i.terapis_id = p_karyawan and i.status = 'selesai' and to_char(coalesce(i.jam_selesai, now()),'YYYY-MM') = p_periode ) hari; $$; grant execute on function hari_kerja_efektif(uuid,text) to authenticated; -- generate_gaji memakai fungsi di atas create or replace function generate_gaji(p_periode text) returns int language plpgsql security definer set search_path = public as $$ declare r record; s skema_gaji; n int := 0; hk int; hari_efektif int; jml_order int; kom numeric; bon numeric; ksb numeric; um numeric; pokok numeric; upah numeric; pot numeric; tunj numeric; hitung numeric; minimal numeric; sesuai numeric; total numeric; v_tipe text; begin for r in select * from karyawan where status = 'aktif' loop s := skema_untuk(r.id); v_tipe := coalesce(r.tipe_gaji, s.tipe_gaji, 'bulanan'); hk := coalesce(s.hari_kerja, 26); hari_efektif := hari_kerja_efektif(r.id, p_periode); select count(*) into jml_order from pesanan_item i where i.terapis_id = r.id and i.status = 'selesai' and to_char(coalesce(i.jam_selesai, now()),'YYYY-MM') = p_periode; select coalesce(sum(nominal),0) into kom from komisi where karyawan_id = r.id and periode = p_periode; select coalesce(sum(nominal),0) into bon from bonus where karyawan_id = r.id and periode = p_periode and status = 'disetujui'; select coalesce(sum(cicilan),0) into ksb from kasbon where karyawan_id = r.id and status = 'disetujui' and sisa > 0; tunj := coalesce(r.tunjangan, s.tunjangan, 0); um := coalesce(s.uang_makan, 0) * hari_efektif; if v_tipe = 'harian' then pokok := 0; upah := coalesce(r.gaji_harian, s.gaji_harian, 0) * hari_efektif; pot := 0; elsif v_tipe = 'training' then pokok := coalesce(r.gaji_pokok, s.gaji_pokok, 0); upah := coalesce(r.gaji_harian, s.gaji_harian, 0) * hari_efektif; pot := greatest(0, coalesce(r.potongan_harian, s.potongan_harian, 0) * (hk - hari_efektif)); else pokok := coalesce(r.gaji_pokok, s.gaji_pokok, 0); upah := 0; pot := case when hari_efektif < hk and hk > 0 then round(pokok / hk) * (hk - hari_efektif) else 0 end; end if; hitung := pokok + upah + tunj + um + kom + bon - pot; minimal := coalesce(s.gaji_minimal, 0); if minimal > 0 and hk > 0 then minimal := round(minimal * least(hari_efektif, hk)::numeric / hk); end if; sesuai := greatest(0, minimal - hitung); total := greatest(hitung, minimal) - ksb; insert into gaji (periode, karyawan_id, tipe_gaji, skema_nama, hari_kerja, hari_hadir, hari_efektif, jml_order, gaji_pokok, upah_harian, tunjangan, uang_makan, total_komisi, total_bonus, potongan_absen, potongan_kasbon, gaji_minimal, penyesuaian, total_terima) values (p_periode, r.id, v_tipe, s.nama, hk, hari_efektif, hari_efektif, jml_order, pokok, upah, tunj, um, kom, bon, pot, ksb, minimal, sesuai, total) on conflict (periode, karyawan_id) do update set tipe_gaji = excluded.tipe_gaji, skema_nama = excluded.skema_nama, hari_hadir = excluded.hari_hadir, hari_efektif = excluded.hari_efektif, jml_order = excluded.jml_order, gaji_pokok = excluded.gaji_pokok, upah_harian = excluded.upah_harian, tunjangan = excluded.tunjangan, uang_makan = excluded.uang_makan, total_komisi = excluded.total_komisi, total_bonus = excluded.total_bonus, potongan_absen = excluded.potongan_absen, potongan_kasbon = excluded.potongan_kasbon, gaji_minimal = excluded.gaji_minimal, penyesuaian = excluded.penyesuaian, total_terima = excluded.total_terima, updated_at = now() where gaji.status = 'draft'; n := n + 1; end loop; return n; end $$; notify pgrst, 'reload schema'; -- ===================================================================== -- PEMERIKSAAN -- ===================================================================== select k.nama, j.nama as jabatan, c.nama as cabang, s.nama as skema, s.tipe_gaji, s.gaji_harian, s.persen_komisi, s.gaji_minimal from karyawan k left join jabatan j on j.id = k.jabatan_id left join cabang c on c.id = k.cabang_id left join lateral (select * from skema_untuk(k.id)) s on true where k.status = 'aktif' order by c.nama, j.nama, k.nama;
